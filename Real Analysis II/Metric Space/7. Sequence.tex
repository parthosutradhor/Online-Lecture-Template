\end{heading}

\vspace{10pt}

\begin{definition}[Sequence in Metric Space]
    Let $(X, d)$ be a metric space. A sequence in $X$ is a function $f: \mathbb{N} \to X$. We usually denote the sequence by $\{x_n\}$, where $x_n = f(n)$ for each $n \in \mathbb{N}$.
\end{definition}
