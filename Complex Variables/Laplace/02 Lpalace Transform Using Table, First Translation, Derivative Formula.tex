\begin{heading}
	Linearity of Laplace Transform
\end{heading}

\vspace{10pt}

\begin{theorem}
	Let $f(t)$ and $g(t)$ be functions such that $\mathcal{L}\{f(t)\}$ and $\mathcal{L}\{g(t)\}$ exist. Then, for any constants $c$, we have
	\[
	\mathcal{L}\{cf(t)\} = c\; \mathcal{L}\{f(t)\},
	\]

	\[
	\mathcal{L}\{f(t) + g(t)\} = \mathcal{L}\{f(t)\} + \mathcal{L}\{g(t)\}.
	\]
\end{theorem}

\begin{heading}
	Important Trigonometric Identities
\end{heading}

\vspace{10pt}

\begin{definition}[Compound Angle Identities]
	\begin{align*}
		\sin(A + B) &= \sin A \cos B + \cos A \sin B, \\
		\sin(A - B) &= \sin A \cos B - \cos A \sin B, \\
		\cos(A + B) &= \cos A \cos B - \sin A \sin B.\\
		\cos(A - B) &= \cos A \cos B + \sin A \sin B.
	\end{align*}
\end{definition}

\vspace{10pt}

\begin{definition}[Product-to-Sum Identities]
	\begin{align*}
		2 \sin A \cos B &= \sin(A+B) + \sin(A-B), \\
		2 \cos A \cos B &= \cos(A+B) + \cos(A-B), \\
		2 \sin A \sin B &= \cos(A-B) - \cos(A+B).
	\end{align*}
\end{definition}

\vspace{10pt}

\begin{definition}[Negative Angle Identities]
	\begin{align*}
		\sin(-A) &= -\sin(A), \\
		\cos(-A) &= \cos(A).
	\end{align*}
\end{definition}
