\begin{heading}
	Infinite Series
\end{heading}

\begin{definition}[Infinite Series]
	Let $\{a_n\}_{n=1}^{\infty}$ be a sequence of real numbers.  
	The infinite series generated by this sequence is the formal expression
	\[
	\sum_{n=1}^{\infty} a_n .
	\]
\end{definition}

\clearpage

\begin{heading}
	Convergence of Infinite Series
\end{heading}

\begin{definition}[Convergence of Infinite Series]
	We define the corresponding sequence of partial sums $(s_m)$ by
	\[
	s_m = a_1 + a_2 + a_3 + \cdots + a_m,
	\]
	and say that the series 
	\(
	\sum_{n=1}^{\infty} a_n
	\)
	converges to $S$ if the sequence $(s_m)$ converges to $S$.  
	In this case, we write
	\[
	\sum_{n=1}^{\infty} a_n = S.
	\]
	\vspace{5pt}
	
	If the sequence $\{S_N\}$ does not converge, the series is called divergent.
\end{definition}

\clearpage

\begin{heading}
	Convergence of Geometric Series
\end{heading}

\vspace{10pt}

\begin{theorem}
	The geometric series 
	\[
	\sum_{k=0}^{\infty} a + ar + ar^{2} + ar^{3} + \cdots
	\]
	converges if and only if $|r| < 1$.
\end{theorem}

\clearpage

\begin{heading}
	Comparison Test
\end{heading}

\vspace{10pt}

\begin{theorem}
	Let $\{a_n\}$ and $\{b_n\}$ be sequences of non-negative real numbers such that  
	\[
	0 \le a_n \le b_n \qquad \text{for all } n.
	\]
	Then:

	\begin{itemize}
		\item[(i)] If $\displaystyle \sum_{n=1}^{\infty} b_n$ converges, then $\displaystyle \sum_{n=1}^{\infty} a_n \text{ also converges}.$

		\item[(ii)] If $\displaystyle \sum_{n=1}^{\infty} a_n$ diverges, then $\displaystyle \sum_{n=1}^{\infty} b_n \text{ also diverges}.$
	\end{itemize}
\end{theorem}

\begin{heading}
	p-series Test
\end{heading}

\vspace{10pt}

\begin{theorem}
	Consider the series
	\[
	\sum_{n=1}^{\infty} \frac{1}{n^{p}},
	\]
	where \(p\) is a real number.

	\begin{itemize}
		\item[(i)] If \(p > 1\), then the series converges.
		\item[(ii)] If \(p \le 1\), then the series diverges.
	\end{itemize}
\end{theorem}

\begin{heading}
	Limit Comparison Test
\end{heading}

\vspace{10pt}

\begin{theorem}
	Let 
	\(
	\sum a_n \text{ and } \sum b_n
	\)
	be two positive–term series.

	\begin{itemize}
		\item[(i)] If 
		\[
		\lim_{n \to \infty} \frac{a_n}{b_n} = L
		\qquad \text{with } 0 < L < \infty,
		\]
		then the series \(\sum a_n\) and \(\sum b_n\) either both converge or both diverge.

		\item[(ii)] If 
		\[
		\lim_{n \to \infty} \frac{a_n}{b_n} = 0
		\quad \text{and } \sum b_n \text{ converges},
		\]
		then \(\sum a_n\) also converges.

		\item[(iii)] If 
		\[
		\lim_{n \to \infty} \frac{a_n}{b_n} = \infty
		\quad \text{and } \sum b_n \text{ diverges},
		\]
		then \(\sum a_n\) also diverges.
	\end{itemize}
\end{theorem}

\begin{heading}
	 D'Alembert’s Ratio Test
\end{heading}

\vspace{10pt}

\begin{theorem}
	Let $\sum a_n$ be a series with positive terms, and suppose the limit
	\[
	L = \lim_{n \to \infty} \frac{a_{n+1}}{a_n}
	\]
	exists (or is infinite). Then:

	\begin{itemize}
		\item[(i)] If \(L < 1\), the series \(\sum a_n\) converges.
		\item[(ii)] If \(L > 1\) (including \(L = \infty\)), the series \(\sum a_n\) diverges.
		\item[(iii)] If \(L = 1\), the test is inconclusive.
	\end{itemize}
\end{theorem}

\begin{heading}
	 Cauchy’s Root Test
\end{heading}

\vspace{10pt}

\begin{theorem}
	Let $\sum a_n$ be a series with non-negative terms, and suppose the limit
	\[
	L = \lim_{n \to \infty} \sqrt[n]{a_n}
	\]
	exists (or is infinite). Then:

	\begin{itemize}
		\item[(i)] If \(L < 1\), the series \(\sum a_n\) converges.
		\item[(ii)] If \(L > 1\) or \(L = \infty\), the series \(\sum a_n\) diverges.
		\item[(iii)] If \(L = 1\), the test is inconclusive.
	\end{itemize}
\end{theorem}

\begin{heading}
	Cauchy Condensation Test
\end{heading}

\vspace{10pt}

\begin{theorem}
Let $\{a_n\}$ be a non-increasing sequence of positive real numbers.  
Then the series
\[
\sum_{n=1}^{\infty} a_n
\]
converges if and only if the series
\[
\sum_{n=1}^{\infty} 2^{n} a_{2^{n}}
\]
converges.
\end{theorem}
