\begin{heading}
	Infinite Series
\end{heading}

\begin{definition}[Infinite Series]
	Let $\{a_n\}_{n=1}^{\infty}$ be a sequence of real numbers.  
	The infinite series generated by this sequence is the formal expression
	\[
	\sum_{n=1}^{\infty} a_n .
	\]
\end{definition}

\clearpage

\begin{heading}
	Convergence of Infinite Series
\end{heading}

\begin{definition}[Convergence of Infinite Series]
	We define the corresponding sequence of partial sums $(s_m)$ by
	\[
	s_m = a_1 + a_2 + a_3 + \cdots + a_m,
	\]
	and say that the series 
	\(
	\sum_{n=1}^{\infty} a_n
	\)
	converges to $S$ if the sequence $(s_m)$ converges to $S$.  
	In this case, we write
	\[
	\sum_{n=1}^{\infty} a_n = S.
	\]
	\vspace{5pt}
	
	If the sequence $\{S_N\}$ does not converge, the series is called divergent.
\end{definition}

\begin{heading}
	Convergence of Geometric Series
\end{heading}

\vspace{10pt}

\begin{theorem}
	The geometric series 
	\[
	1 + x + x^{2} + x^{3} + \cdots \ \text{to } \infty
	\]
	satisfies the following:

	\begin{itemize}
		\item it converges if $-1 < x < 1$, i.e.\ if $|x| < 1$;
		\item it diverges if $x \ge 1$;
		\item it oscillates finitely if $x = -1$;
		\item it oscillates infinitely if $x < -1$.
	\end{itemize}
\end{theorem}

\begin{heading}
	Comparison Test
\end{heading}

\vspace{10pt}

\begin{theorem}[Comparison Test]
	Let $\{a_n\}$ and $\{b_n\}$ be sequences of non-negative real numbers such that  
	\[
	0 \le a_n \le b_n \qquad \text{for all } n.
	\]
	Then:

	\begin{itemize}
		\item[(i)] If $\displaystyle \sum_{n=1}^{\infty} b_n$ converges, then  
		\[
		\sum_{n=1}^{\infty} a_n \quad \text{also converges}.
		\]

		\item[(ii)] If $\displaystyle \sum_{n=1}^{\infty} a_n$ diverges, then  
		\[
		\sum_{n=1}^{\infty} b_n \quad \text{also diverges}.
		\]
	\end{itemize}
\end{theorem}
